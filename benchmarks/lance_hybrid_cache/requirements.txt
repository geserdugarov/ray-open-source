# pylance is installed separately from $HOME/git/lance-open-source. Use a local
# checkout with the distributed-cache and prewarm feature; the
# `Session.with_distributed_cache`, `Session.invalidate_index_cache`,
# `Session.size_bytes`, and the strict
# `dataset.prewarm_index(name, partition_ids=...)` path are all required by the
# scenarios + drivers in this directory. The compatible build preserves this
# Python API exactly — the public-API diff vs the `private-cache-7.0-ver-1`
# line is additive only — so the scenarios + drivers need no source changes. The feature
# is baked into the local wheel and a plain editable install is sufficient:
#   pip install -e "$HOME/git/lance-open-source/python"
# Re-running `pip install -e` (or `maturin develop --release`) is only
# required after Rust changes. The sharded-mode actor paths additionally
# require pylance to expose `dataset.compute_partition_ids` and
# `dataset.search_partitions`; without those wrappers the actor raises a
# clear RuntimeError on first use.

# Must stay inside the pylance wheel's own lance-namespace pin. Two build lines
# exist: the open-source build (pyproject.toml: `lance-namespace>=0.7.7,<0.8`,
# resolved to final `0.7.7`) and the internal cluster build, whose wheel pins an
# exact PRE-RELEASE local version `lance-namespace==0.7.7rc0+h0.cbu.mrs.370.r1`.
# Because `0.7.7rc0 < 0.7.7`, a `>=0.7.7` floor EXCLUDES that pre-release, so
# `pip install --no-index pylance ... -r requirements.txt` fails with
# ResolutionImpossible on the cluster. Lowering the floor to `>=0.7.7rc0` admits
# it (the `rc0` mention also turns on pip pre-release handling) while final
# `0.7.7 >= 0.7.7rc0` still satisfies it, so the open-source path is unaffected —
# pip prefers final `0.7.7` when present, and the internal pylance's exact `==`
# pin selects the rc0+local wheel only when that wheel is staged in the
# wheelhouse. Keep the `<0.8` ceiling: a looser pin lets `pip wheel` stage
# lance-namespace 0.8.x, which violates the pylance wheel's upper bound and makes
# `--no-index` silently backtrack to an older pylance with no
# `dataset.prewarm_index(partition_ids=...)`. The rc0+local lance-namespace wheel
# is NOT on PyPI — the cluster wheelhouse must carry it alongside the pylance wheel.
lance-namespace>=0.7.7rc0,<0.8
